#include <tunables/global>

# Housekeeper Agent: one Python process (no s6, no shell) that
#   - makes outbound HTTP/websocket calls (Supervisor proxy, the Housekeeper service, an optional
#     health-check URL) and DNS lookups,
#   - reads its own code and /data/options.json, and reads/writes /data/memory.json.
# Anything else (shells, other binaries, writes outside /data, capabilities, raw sockets) is denied.
# The Supervisor renames this profile to the installed App's slug when it loads it.
profile housekeeper_agent flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Network: TCP for HTTP(S)/websocket, UDP for DNS. No raw/packet sockets.
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,

  # Stop/restart/update: the Supervisor stops the App with SIGTERM (SIGKILL after the timeout).
  signal (receive) set=(term, int, kill),

  # The Python interpreter and its libraries (python:*-alpine installs under /usr/local).
  /usr/local/bin/python* rix,
  /usr/local/ r,
  /usr/local/** r,
  /usr/local/lib/** rm,
  /lib/** rm,
  /usr/lib/** rm,

  # The App's own code (read-only; the image precompiles it and disables bytecode writes).
  /app/ r,
  /app/** r,

  # Settings (written by the Supervisor) and the agent's memory file (atomic write via *.tmp).
  /data/ r,
  /data/** rw,

  # Name resolution, TLS trust store, time zone, user lookup.
  /etc/hosts r,
  /etc/hostname r,
  /etc/resolv.conf r,
  /etc/nsswitch.conf r,
  /etc/services r,
  /etc/protocols r,
  /etc/passwd r,
  /etc/group r,
  /etc/localtime r,
  /etc/ssl/** r,
  /usr/share/ca-certificates/** r,
  /usr/share/zoneinfo/** r,

  # Python runtime housekeeping.
  /dev/null rw,
  /dev/urandom r,
  /etc/ld-musl-*.path r,
  @{PROC}/sys/kernel/random/uuid r,
  @{PROC}/sys/vm/overcommit_memory r,
  @{PROC}/@{pid}/exe r,
  @{PROC}/@{pid}/fd/ r,
  @{PROC}/@{pid}/mounts r,
  @{PROC}/@{pid}/status r,
  @{PROC}/sys/kernel/osrelease r,
}
